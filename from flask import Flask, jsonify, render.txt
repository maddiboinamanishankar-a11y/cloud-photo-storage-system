# ============================================================
# CLOUD SERVER
# ============================================================

@dataclass
class Server:
    id: int
    cpu: float
    memory: float
    network: float
    requests: int
    status: str = "HEALTHY"
    cost_per_hour: float = 0.10

    def update(self):
        # Simulate changing cloud workload
        self.cpu = max(
            0,
            min(100, self.cpu + random.uniform(-15, 18))
        )

        self.memory = max(
            0,
            min(100, self.memory + random.uniform(-10, 12))
        )

        self.network = max(
            0,
            min(100, self.network + random.uniform(-15, 15))
        )

        self.requests = random.randint(100, 3000)

        # Health detection
        if self.cpu >= 90 or self.memory >= 90:
            self.status = "CRITICAL"

        elif self.cpu >= 75 or self.memory >= 75:
            self.status = "WARNING"

        else:
            self.status = "HEALTHY"


# ============================================================
# CLOUD OPTIMIZER
# ============================================================

class CloudOptimizer:

    def __init__(self):

        self.servers = [
            Server(
                id=1,
                cpu=45,
                memory=40,
                network=30,
                requests=500
            ),

            Server(
                id=2,
                cpu=50,
                memory=45,
                network=35,
                requests=700
            ),

            Server(
                id=3,
                cpu=35,
                memory=50,
                network=25,
                requests=400
            )
        ]

        self.history = []

        self.total_cost = 0

        self.events = []

    # --------------------------------------------------------
    # MONITORING
    # --------------------------------------------------------

    def monitor(self):

        for server in self.servers:
            server.update()

        cpu = self.average_cpu()
        memory = self.average_memory()

        self.history.append({
            "time": time.strftime("%H:%M:%S"),
            "cpu": cpu,
            "memory": memory,
            "servers": len(self.servers)
        })

        # Keep last 50 records
        self.history = self.history[-50:]

    # --------------------------------------------------------
    # AVERAGE CPU
    # --------------------------------------------------------

    def average_cpu(self):

        if not self.servers:
            return 0

        return round(
            sum(s.cpu for s in self.servers)
            / len(self.servers),
            2
        )

    # --------------------------------------------------------
    # AVERAGE MEMORY
    # --------------------------------------------------------

    def average_memory(self):

        if not self.servers:
            return 0

        return round(
            sum(s.memory for s in self.servers)
            / len(self.servers),
            2
        )

    # --------------------------------------------------------
    # AVERAGE NETWORK
    # --------------------------------------------------------

    def average_network(self):

        if not self.servers:
            return 0

        return round(
            sum(s.network for s in self.servers)
            / len(self.servers),
            2
        )

    # --------------------------------------------------------
    # TOTAL REQUESTS
    # --------------------------------------------------------

    def total_requests(self):

        return sum(
            server.requests
            for server in self.servers
        )

    # --------------------------------------------------------
    # AUTO SCALING
    # --------------------------------------------------------

    def auto_scale(self):

        cpu = self.average_cpu()
        memory = self.average_memory()

        # SCALE UP
        if cpu > 75 or memory > 80:

            if len(self.servers) < 10:

                new_id = max(
                    server.id
                    for server in self.servers
                ) + 1

                new_server = Server(
                    id=new_id,
                    cpu=random.uniform(10, 30),
                    memory=random.uniform(20, 40),
                    network=random.uniform(10, 30),
                    requests=random.randint(100, 400)
                )

                self.servers.append(new_server)

                event = (
                    f"Scale UP: Server {new_id} "
                    f"added because workload is high."
                )

                self.events.append({
                    "time": time.strftime("%H:%M:%S"),
                    "message": event
                })

                return event

        # SCALE DOWN
        elif cpu < 30 and memory < 35:

            if len(self.servers) > 1:

                removed = self.servers.pop()

                event = (
                    f"Scale DOWN: Server "
                    f"{removed.id} removed."
                )

                self.events.append({
                    "time": time.strftime("%H:%M:%S"),
                    "message": event
                })

                return event

        return "No scaling required."

    # --------------------------------------------------------
    # WORKLOAD PREDICTION
    # --------------------------------------------------------

    def predict(self):

        if len(self.history) < 5:
            return "Collecting data..."

        recent = self.history[-5:]

        old_cpu = recent[0]["cpu"]
        new_cpu = recent[-1]["cpu"]

        trend = new_cpu - old_cpu

        if trend > 15:
            return "HIGH LOAD EXPECTED"

        elif trend > 5:
            return "LOAD INCREASING"

        elif trend < -15:
            return "LOW LOAD EXPECTED"

        elif trend < -5:
            return "LOAD DECREASING"

        return "STABLE"

    # --------------------------------------------------------
    # COST CALCULATION
    # --------------------------------------------------------

    def calculate_cost(self):

        hourly_cost = sum(
            server.cost_per_hour
            for server in self.servers
        )

        # Simulation: each update = 1 minute
        minute_cost = hourly_cost / 60

        self.total_cost += minute_cost

        return round(self.total_cost, 4)

    # --------------------------------------------------------
    # HEALTH MONITOR
    # --------------------------------------------------------

    def health(self):

        critical = [
            server.id
            for server in self.servers
            if server.status == "CRITICAL"
        ]

        warning = [
            server.id
            for server in self.servers
            if server.status == "WARNING"
        ]

        if critical:
            overall = "CRITICAL"

        elif warning:
            overall = "WARNING"

        else:
            overall = "HEALTHY"

        return {
            "overall": overall,
            "critical": critical,
            "warning": warning
        }

    # --------------------------------------------------------
    # OPTIMIZATION
    # --------------------------------------------------------

    def optimize(self):

        scaling = self.auto_scale()

        prediction = self.predict()

        cost = self.calculate_cost()

        return {
            "scaling": scaling,
            "prediction": prediction,
            "cost": cost
        }

    # --------------------------------------------------------
    # STATUS
    # --------------------------------------------------------

    def get_status(self):

        health = self.health()

        return {
            "servers": [
                asdict(server)
                for server in self.servers
            ],

            "server_count":
                len(self.servers),

            "cpu":
                self.average_cpu(),

            "memory":
                self.average_memory(),

            "network":
                self.average_network(),

            "requests":
                self.total_requests(),

            "cost":
                round(self.total_cost, 4),

            "prediction":
                self.predict(),

            "health":
                health,

            "history":
                self.history,

            "events":
                self.events[-10:]
        }
